---
import EntityPillar from "../_shared/_EntityPillar.astro";
import { getCollection } from "astro:content";
import { buildPillarGraph } from "../../lib/schema";
import { resolveRoute } from "../../i18n/routes";
import { siteConfig } from "../../config/site.config";
import type { Locale } from "../../config/site.config";
const locale = (Astro.currentLocale || "en") as Locale;
const entries = (await getCollection("evidence")).filter((e: any) => e.data.locale === locale).sort((a: any, b: any) => (a.data.order || 0) - (b.data.order || 0));
const listUrl = siteConfig.url + resolveRoute(locale, "/evidence/");
const pageGraph = buildPillarGraph({
  name: "Titanium Engineering Evidence & Data",
  description: "Structured technical data, cutting parameters, tolerances, surface roughness, and material properties for titanium manufacturing.",
  url: listUrl,
  collectionKey: "evidence",
  locale,
  items: entries.map((e: any) => ({
    name: e.data.title,
    description: e.data.description,
    url: siteConfig.url + resolveRoute(locale, "/evidence/" + e.id + "/"),
  })),
}, siteConfig);
---
<EntityPillar collectionName="evidence" pageTitle="Titanium Engineering Evidence & Data" pageDescription="Structured technical data, cutting parameters, tolerances, surface roughness, and material properties for titanium manufacturing." locale={locale} path="/evidence/" pageGraph={pageGraph} />
